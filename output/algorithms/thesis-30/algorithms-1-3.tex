\begin{algorithm}[tbp]
\caption{Object-centric discrete-event simulation}
\label{alg:object-centric-discrete-event-simulation}
\small
\begin{algorithmic}[1]
\Require Model $M$, configuration $K$, discovered parameters $D$, random generator $R$
\Ensure State $S$ containing objects, completed events, pending work, and metrics
\State $S \gets$ supplied state or empty state; $t \gets K.\mathrm{startTime}$; $idle \gets 0$
\State Initialise clocks, start-activity set, and wall-clock runtime timer
\State $d \gets t + K.\mathrm{maxSimTime}$ if configured, otherwise $+\infty$
\While{no user stop and no non-time stopping limit is reached}
    \While{a scheduled completion exists at or before $t$}
        \State Remove the earliest instance and \Call{Complete}{$S$, instance}
        \If{a non-time stopping limit is reached} \State \textbf{break} \EndIf
    \EndWhile
    \If{a stopping limit is reached or $t \ge d$} \State \textbf{break} \EndIf
    \State $initial \gets$ no completed events and no running instances
    \State $C \gets \Call{GenerateCandidates}{M,S,D}$ \Comment{includes ready obligations}
    \State $C \gets \Call{OrderCandidates}{C,S,D,R}$
    \State $catchup \gets \emptyset$
    \ForAll{$c \in C$ in this order}
        \If{$initial$ and $c$ is not a configured start activity} \State \textbf{continue} \EndIf
        \If{start policy, concurrency limit, or activity calendar blocks $c$}
            \State \textbf{continue}
        \EndIf
        \If{a participant of $c$ is busy}
            \State $c \gets$ first admissible alternative binding, preserving required objects
            \If{no such binding exists} \State \textbf{continue} \EndIf
        \EndIf
        \State Attempt \Call{AdmitAndSchedule}{$c,M,S,K,R$}
        \If{a successful start advances an arrival schedule but it remains due}
            \State Add its activity to $catchup$
        \EndIf
    \EndFor
    \State $F \gets \emptyset$; add the earliest scheduled completion time if work is running
    \State Add future arrival times whose start policy and concurrency permit arrival to $F$
    \State Add $t$ for eligible activities in $catchup$ whose calendar is open
    \If{some otherwise eligible activity has a closed configured calendar}
        \State Add the next hourly calendar boundary to $F$
    \EndIf
    \If{$F=\emptyset$} \State \textbf{break} \EndIf
    \If{$d<+\infty$} \State $F \gets F\cup\{d\}$ \EndIf
    \State $idle \gets 0$ if work is running, otherwise $idle+1$
    \If{$idle>1000$} \State \textbf{break} \EndIf
    \State $t \gets \min F$; $S.\mathrm{currentTime} \gets t$
\EndWhile
\State \Return $S$
\end{algorithmic}
\end{algorithm}


\begin{algorithm}[tbp]
\caption{Activity timing and service-time estimation}
\label{alg:activity-timing-and-service-time-estimation}
\small
\begin{algorithmic}[1]
\Require Normalised OCEL $L$, optional anchors $H$, mode $m\in\{\mathrm{minimum},\mathrm{p25},\mathrm{p50},\mathrm{mean}\}$ (default $\mathrm{p25}$)
\Ensure Per-activity timing summaries and estimated service-distribution parameters
\State Build object timelines in stable timestamp order
\State Initialise empty per-activity sample buckets $B$
\ForAll{events $e$ with activity $a$, valid timestamp $t$, and participating objects}
    \State $V \gets$ valid immediately preceding timestamps on the participating object timelines
    \State $N \gets$ valid immediately following timestamps on these timelines
    \ForAll{$u\in N$} \State Append $\max(0,u-t)$ in seconds to $B[a].\mathrm{forward}$ \EndFor
    \If{$V=\emptyset$}
        \State Append $0$ to $B[a].\mathrm{flow}$ and $B[a].\mathrm{sojourn}$
        \State \textbf{continue}
    \EndIf
    \State $early\gets\min V$; $late\gets\max V$
    \State Append $late-early$, $t-early$, $t-late$ in seconds to sync, flow, and sojourn buckets
    \State Group $V$ by object type; record each group's earliest and latest timestamps
    \State Append $\max(\mathrm{typeLatest})-\min(\mathrm{typeEarliest})$ to pooling
    \State Append $\max(\mathrm{typeLatest})-late$ to lagging \Comment{equals zero here}
\EndFor
\ForAll{activities $a$ present in $B$ or $H$}
    \State $D\gets B[a].\mathrm{forward}$; $Z\gets$ positive values of $B[a].\mathrm{sojourn}$
    \If{$D\ne\emptyset$ and $Z\ne\emptyset$}
        \State $X\gets D$ if $\mathrm{upperMedian}(D)\le\mathrm{upperMedian}(Z)$, otherwise $Z$
    \Else
        \State $X\gets D$ if nonempty, else $Z$ if nonempty, else $B[a].\mathrm{sojourn}$
    \EndIf
    \If{an anchor for $a$ is supplied}
        \State Use supplied service parameters, with the implementation's defaults for missing fields
    \ElsIf{$X\ne\emptyset$}
        \State Sort $X$ increasingly; $n\gets |X|$
        \State $k\gets\begin{cases}1 & m=\mathrm{minimum}\\ \lceil n/4\rceil & m=\mathrm{p25}\\ \lceil n/2\rceil & m=\mathrm{p50}\\ n & m=\mathrm{mean}\end{cases}$
        \State $Y\gets$ first $k$ values of $X$
        \State Service parameters $\gets \mathrm{mean}(Y),\mathrm{populationStd}(Y),\min Y,\max Y$
    \Else
        \State Use the sojourn summary; an empty sample has mean/std/min $0$ and no maximum
    \EndIf
    \State Summarise timing buckets using population standard deviations
    \State $waitingMean\gets\max(0,sojournMean-serviceMean)$ when defined, otherwise undefined
    \State Emit timing summaries and service parameters with default distribution type lognormal
\EndFor
\State \Return all emitted summaries
\end{algorithmic}
\end{algorithm}


\begin{algorithm}[tbp]
\caption{Object creation and deactivation discovery}
\label{alg:object-initialization-exit-discovery}
\begin{algorithmic}[1]
\Require Normalised OCEL $L$, object type $T$, threshold $h\in(0,1]$ (default $0.9$)
\Ensure Creating-activity set $C$ and deactivating-activity set $D$ for $T$
\State $O_T\gets$ all declared objects of type $T$; $n\gets|O_T|$
\If{$n=0$} \State \Return $\emptyset,\emptyset$ \EndIf
\State Initialise first-activity and last-activity counters $F$ and $Z$ to zero
\ForAll{$o\in O_T$}
    \State $E_o\gets$ events involving $o$ that have an activity name
    \If{$E_o=\emptyset$} \State \textbf{continue} \EndIf
    \State Validate timestamps and normalise to UTC, treating naive timestamps as UTC
    \State $e_{first}\gets\arg\min_{e\in E_o}(\mathrm{timestamp}(e),\mathrm{stringID}(e))$
    \State $e_{last}\gets\arg\max_{e\in E_o}(\mathrm{timestamp}(e),\mathrm{stringID}(e))$
    \State Increment $F[\mathrm{activity}(e_{first})]$ and $Z[\mathrm{activity}(e_{last})]$
\EndFor
\State $C\gets\{a\mid F[a]/n\ge h\}$
\State $D\gets\{a\mid Z[a]/n\ge h\}$ \Comment{independent of $C$}
\State \Return $C,D$
\end{algorithmic}
\end{algorithm}
